  function pills(el, list, curId, cb, withRange) {

      html += '<button type="button" class="calc-pill' + (it.custom ? ' custom' : '') + (it.id === curId ? ' on' : '') +

  /* ===== Расчёт ===== */
  function farePerOrder() {
    if (S.manual) return Math.max(0, S.avgCheck);
    var extraKm = Math.max(0, S.avgKm - S.tIk), extraMin = Math.max(0, S.avgMin - S.tIm);
    return Math.max(S.tMin, S.tMin + extraKm * S.tKm + extraMin * S.tMn);
  }

  function calc(st) {
    var fare = farePerOrder.call ? (function () {
      if (st.manual) return Math.max(0, st.avgCheck);
      var ek = Math.max(0, st.avgKm - st.tIk), em = Math.max(0, st.avgMin - st.tIm);
      return Math.max(st.tMin, st.tMin + ek * st.tKm + em * st.tMn);
    })() : 0;
    var ordersDay = st.opHours * st.opOrders;
    var grossDay = fare * ordersDay * (1 + st.opBoost / 100);
    var daysMonth = st.opShifts * 4.345;
    var grossMonth = grossDay * daysMonth;
    var yaFee = grossMonth * st.tYa / 100;
    var parkFee = grossMonth * st.parkFee / 100;
    var wdFee = Math.max(0, grossMonth - yaFee - parkFee) * st.wdFee / 100;
    var fuelMonth = st.mileage * st.fCons / 100 * st.fPrice * daysMonth;
    var rentMonth = st.rent === 'own' ? 0 : (st.rentCalendar ? st.rPrice * 30.44 : st.rPrice * daysMonth);
    var miscMonth = st.misc;
    var tax = 0, taxNote = '';
    if (st.tax === 'usn') {
      var contrib = USN_FIXED / 12 + Math.max(0, (grossMonth * 12 - 300000) * 0.01) / 12;
      var usn6 = grossMonth * 0.06;
      var payable = Math.max(0, usn6 - contrib);
      tax = payable + contrib;
      taxNote = 'УСН 6%: налог ' + fmt(usn6) + ' уменьшен на взносы ' + fmt(contrib) + ' (ИП без работников может уменьшать налог на взносы вплоть до нуля). К доплате: ' + fmt(payable) + ' налога + ' + fmt(contrib) + ' взносов.';
    } else {
      var physRate = st.deductUsed ? 4 : 3, jurRate = st.deductUsed ? 6 : 4;
      var rate = (physRate * (100 - st.jurShare) + jurRate * st.jurShare) / 100;
      tax = grossMonth * rate / 100;
      taxNote = 'НПД ' + (st.deductUsed ? '4%/6%' : '3%/4% с учётом вычета 10 000 ₽') +
        ': средняя ставка ' + rate.toFixed(2).replace('.', ',') + '% при доле заказов от юрлиц ' + st.jurShare + '%.';
      if (st.tax === 'ipnpd') taxNote += ' ИП на НПД не платит фиксированные взносы.';
    }
    var net = grossMonth - yaFee - parkFee - wdFee - fuelMonth - rentMonth - miscMonth - tax;
    return {fare: fare, ordersDay: ordersDay, grossDay: grossDay, daysMonth: daysMonth, grossMonth: grossMonth,
      yaFee: yaFee, parkFee: parkFee, wdFee: wdFee, fuelMonth: fuelMonth, rentMonth: rentMonth,
      miscMonth: miscMonth, tax: tax, taxNote: taxNote, net: net,
      netDay: daysMonth ? net / daysMonth : 0, netHour: (daysMonth && st.opHours) ? net / (daysMonth * st.opHours) : 0};
  }

  function saveHash() {

  function loadHash() {
    try {
      var m = location.hash.match(/#c=([A-Za-z0-9+/=]+)/);
      if (!m) return;
      var o = JSON.parse(decodeURIComponent(escape(atob(m[1]))));
      Object.keys(DEF).forEach(function (k) { if (o[k] !== undefined) S[k] = o[k]; });
    } catch (e) {}
  }

<div class="step reveal" id="taxes">
    <div class="step-head"><h2>Налоги водителя: НПД, вычет 10 000 ₽ и ИП</h2></div>
    <div class="step-text">
<p>Работать в такси легально можно тремя основными способами, и налог считается по-разному. Разберём каждый — так, чтобы вы могли объяснить разницу коллеге на парковке.</p>
<h3>Самозанятость (НПД): 4% и 6%</h3>
<p>Самозанятый платит налог на профессиональный доход: <b>4% с заказов от физических лиц</b> и <b>6% с заказов от юридических лиц</b> (когда поездку оплачивает компания пассажира). Обязательных страховых взносов нет, лимит — 2,4 млн ₽ дохода в год. Ставки зафиксированы законом до конца 2028 года. Важный нюанс, о котором часто забывают: на НПД <b>не идёт пенсионный стаж</b> — взносы в Социальный фонд можно платить добровольно через «Мой налог».</p>
<h3>Вычет 10 000 ₽: что это и почему о нём стоит помнить</h3>
<p>Каждому новому самозанятому государство один раз в жизни даёт налоговый вычет 10 000 ₽. Механика такая: пока вычет не исчерпан, налог начисляется по сниженным ставкам — <b>3% вместо 4%</b> с физлиц и <b>4% вместо 6%</b> с юрлиц, а недостающий процент «доплачивает» вычет. Иными словами, первые 10 000 ₽ вашего налога государство платит за вас. При обороте 100 000 ₽ в месяц вычет растянется примерно на десять месяцев, при 250 000 ₽ — примерно на четыре. Когда вычет кончится, ставка просто станет стандартной — никаких действий от вас не требуется. Проверить остаток вычета можно в приложении «Мой налог». В калькуляторе выше это переключатель «вычет уже исчерпан» — если вы работаете недавно и галочку не ставили, расчёт использует ставки 3%/4%.</p>
<h3>ИП: НПД или УСН 6%</h3>
<p>Индивидуальный предприниматель тоже может платить НПД (те же 4%/6% и тот же вычет) — и тогда фиксированных взносов нет. Классический вариант — УСН «доходы»: <b>6% с оборота</b>, плюс обязательные страховые взносы — в 2026 году это <b>57 390 ₽ в год</b> и ещё 1% с дохода свыше 300 000 ₽. При этом ИП без работников уменьшает налог на сумму взносов вплоть до нуля, поэтому реальная нагрузка УСН часто близка к 6% «всё включено». Плюс УСН — пенсионный стаж и взносы идут автоматически; минус — взносы платятся даже без дохода.</p>

Часто выигрывает аренда на старте (пока нет своей машины под требования) и своё авто вдолгую.</p>

        function renderSearch(query) {
            const q = query.trim().toLowerCase();
            const hits = q ? SEARCH_INDEX.filter(i => (i.title + ' ' + i.keys).toLowerCase().includes(q)) : SEARCH_INDEX.slice(0, 6);
            searchResults.innerHTML = hits.length
                ? hits.map(h => `<a class="search-hit" href="${h.url}"><span class="tag ${h.cls}">${h.tag}</span>${h.title}</a>`).join('')
                : '<div class="search-none">Ничего не найдено. Попробуйте другое слово.</div>';
        }
